cmake_minimum_required(VERSION 3.20)
project(helios VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(OpenMP QUIET)
find_package(MPI QUIET COMPONENTS CXX)
find_package(Threads REQUIRED)
add_library(helios_core src/core.cpp)
target_include_directories(helios_core PUBLIC include)
if(MSVC)
  target_compile_options(helios_core PRIVATE /W4)
else()
  target_compile_options(helios_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(OpenMP_CXX_FOUND)
  target_link_libraries(helios_core PUBLIC OpenMP::OpenMP_CXX)
  target_compile_definitions(helios_core PUBLIC HELIOS_OPENMP=1)
endif()
add_executable(helios src/main.cpp)
target_link_libraries(helios PRIVATE helios_core)
target_link_libraries(helios PRIVATE Threads::Threads)
if(MPI_CXX_FOUND)
  target_link_libraries(helios PRIVATE MPI::MPI_CXX)
  target_compile_definitions(helios PRIVATE HELIOS_MPI=1)
endif()
if(BUILD_TESTING)
  add_executable(helios_tests tests/tests.cpp)
  target_link_libraries(helios_tests PRIVATE helios_core)
  add_test(NAME core COMMAND helios_tests)
  add_test(NAME cli_help COMMAND helios --help)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    set(VALIDATION_ARGS --exe $<TARGET_FILE:helios>)
    if(OpenMP_CXX_FOUND)
      list(APPEND VALIDATION_ARGS --openmp)
    endif()
    if(MPI_CXX_FOUND)
      list(APPEND VALIDATION_ARGS --mpi-exe $<TARGET_FILE:helios> --mpiexec ${MPIEXEC_EXECUTABLE})
    endif()
    add_test(NAME backend_validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/validate_backends.py ${VALIDATION_ARGS})
    set_tests_properties(backend_validation PROPERTIES TIMEOUT 90)
    add_test(NAME bvh_validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/validate_bvh.py --exe $<TARGET_FILE:helios>)
    set_tests_properties(bvh_validation PROPERTIES TIMEOUT 90)
    add_test(NAME cli_validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/validate_cli.py --exe $<TARGET_FILE:helios>)
    set_tests_properties(cli_validation PROPERTIES TIMEOUT 30)
    add_test(NAME benchmark_analysis COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_benchmark_analysis.py)
    set_tests_properties(benchmark_analysis PROPERTIES TIMEOUT 30)
  endif()
endif()
